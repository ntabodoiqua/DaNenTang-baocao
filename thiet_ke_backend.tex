\paragraph{Bảng USER - Người dùng}
\textbf{Mục đích:} Lưu trữ thông tin tài khoản và hồ sơ người dùng hệ thống.

\hline
\endfirsthead
\hline
\textbf{Tên cột} & \textbf{Kiểu dữ liệu} & \textbf{Ràng buộc} & \textbf{Mô tả} \\
\hline
\endhead
id & BIGINT & PRIMARY KEY, AUTO\_INCREMENT & Mã định danh duy nhất của người dùng \\
\hline

\hline
password & VARCHAR(255) & NOT NULL & Mật khẩu đã được mã hóa (BCrypt) \\
\hline

\hline
phone & VARCHAR(20) & UNIQUE & Số điện thoại liên hệ \\
\hline

\hline

\hline

\hline

\hline

\hline

\hline
